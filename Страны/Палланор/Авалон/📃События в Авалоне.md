```dataviewjs
// Указываем фильтры (можно оставить пустым, чтобы показывать все)
let filterCity = dv.current().Город ?? null;       // Например: "[[🏙️Авалон]]"
let filterPlayers = dv.current().Игроки ?? null;   // Например: "[[🚩Пачаток]]"

let pages = dv.pages("#Событие");

// Словарь для группировки по дате
let grouped = {};

// Проходим по каждой заметке
for (let p of pages) {
    // Фильтруем по городу
    if (filterCity && p.Город?.path !== filterCity?.path) continue;

    // Фильтруем по игрокам
    if (filterPlayers && p.Игроки?.path !== filterPlayers?.path) continue;

    let content = await dv.io.load(p.file.path);
    
    // Убираем хэштег #Событие и мета-блоки
    content = content.replace(/#Событие/g, "");
    content = content.replace(/---.*?---/gs, ""); 
    
    let gameDate = p["Игровая дата"] ?? "Дата не указана";
    
    // Добавляем ссылку на заметку
    let link = `[[${p.file.name}]]`;
    content = `${link}: ${content.trim()}`;
    
    if (!grouped[gameDate]) grouped[gameDate] = [];
    grouped[gameDate].push(content);
}

// Сортируем даты
let sortedDates = Object.keys(grouped).sort();

// Выводим события по группам в виде списка
for (let date of sortedDates) {
    dv.paragraph(`**${date}:**`);
    let items = grouped[date].map(e => `- ${e}`).join("\n");
    dv.paragraph(items);
}
```
